'''
settings = {
    "background":   "#05080D",
    "border":       "#0E2A35",

    "console-font": pygame.font.SysFont("Consolas", 25),
    "console-i":    "#FF2E60", # command icon
    "console-c":    "#FFFF60", # command text
    "console-t":    "#4FD3E8", # title
    "console-l":    "#6FA8B8", # label
    "console-v":    "#D8F3FF", # value
}


def console_command_info():
    return []


def get_octa_points(center, radius):
    points = []
    cx, cy = center
    
    angle_step = math.pi / 4     
    for i in range(8):
        angle = i * angle_step
        x = cx + radius * math.cos(angle)
        y = cy + radius * math.sin(angle)
        points.append([x, y])
        
    return points

console_info = { 
    "lines": [ "$<i$>" ], 
    "commands": {
        "info": console_command_info
    } 
}
console_tag  = re.compile(r"\$([<^>])([cvlti])\$")

flux1_grid = [
    0, 1, 0, 0,
    0, 0, 1, 0,
    1, 1, 1, 0,
    0, 0, 0, 0
]

flux2_grid = [
    1, 0, 0, 0,
    1, 0, 1, 0,
    1, 1, 0, 0,
    0, 0, 0, 0
]

last_step = pygame.time.get_ticks()

running = True
while running:
    window.fill(settings["background"])

    # outside border
    outside = pygame.draw.rect(window, settings["border"], (10, 30, 1280-20, 720-60), 5)

    # console
    console = pygame.draw.rect(window, settings["border"], (outside.x + 10, outside.y + 10, (outside.w - 20) // 4, (outside.h - 20) * (2/3)), 5)

    #$<t$val"
    x, y = 40, 40
    for raw_line in console_info["lines"]:
        # Create ["", "<", "i", ">", ...]
        parts = console_tag.split(raw_line)

        # Group to [["<", "i", ">"], ["<", "c", "info"]]
        line = []
        for i in range(1, len(parts), 3):
            line.append([parts[i], parts[i+1], parts[i+2]])

        x = 40
        max_height = 0
        for align, color, text in line:
            text_surf = settings["console-font"].render(text, True, settings["console-"+color])

            max_height = max(text_surf.get_height(), max_height)

            match align:
                case "^":
                    x = (console.w - text_surf.get_width()) // 2
                case ">":
                    x = (console.w - 10 - 40 - text_surf.get_width())

            if color == "i": # icon is treated as seperate
                x = 40 - text_surf.get_width()

            window.blit(text_surf, (console.x+x, console.y+y))
            x += text_surf.get_width() + 10

        y += max_height

    # flux
    flux = pygame.draw.rect(window, settings["border"], (outside.x + 10, console.h + console.y + 5, (outside.w - 20) // 4, (outside.h - 30) * (1/3)), 5)
    
    flux1_points = get_octa_points(
        (flux.x + (flux.width // 4), flux.y + (flux.height // 2)),
        flux.height * 0.3
    )
    pygame.draw.polygon(window, settings['border'], flux1_points, width=2)

    now = pygame.time.get_ticks()
    step_due = (now - last_step) >= 250
    if step_due:
        last_step = now

    snap = list(flux1_grid)
    for idx, point in enumerate(snap):
        col = idx % 4
        row = idx // 4

        # Drawing logic remains exactly the same
        pygame.draw.rect(
            window, 
            settings["border"], 
            [flux.x + 10 + (col * 10), flux.y + 10 + (row * 10), 10, 10], 
            width=2 if point == 0 else 0
        )

        # Count live neighbours, wrapping around the edges (torus)
        alive = 0
        for d_row in (-1, 0, 1):
            for d_col in (-1, 0, 1):
                if d_row == 0 and d_col == 0:
                    continue
                n_row = (row + d_row) % 4
                n_col = (col + d_col) % 4
                alive += snap[n_row * 4 + n_col]

        # --- CRITICAL CHANGE: Explicitly update 'flux1_grid' for all cases ---
        if step_due:
            if point == 1:
                flux1_grid[idx] = 1 if alive in (2, 3) else 0
            else:
                flux1_grid[idx] = 1 if alive == 3 else 0

    flux2_points = get_octa_points(
        (flux.x + flux.w + (-flux.height * 0.4), flux.y + (flux.height // 2)),
        flux.height * 0.3
    )
    pygame.draw.polygon(window, settings['border'], flux2_points, width=2)

    snap = list(flux2_grid)
    for idx, point in enumerate(snap):
        col = idx % 4
        row = idx // 4

        # Drawing logic remains exactly the same
        pygame.draw.rect(
            window, 
            settings["border"], 
            [flux.x + flux.w + (-flux.height * 0.4) + 10 + (col * 10), flux.y + 10 + (row * 10), 10, 10], 
            width=2 if point == 0 else 0
        )

        # Count live neighbours, wrapping around the edges (torus)
        alive = 0
        for d_row in (-1, 0, 1):
            for d_col in (-1, 0, 1):
                if d_row == 0 and d_col == 0:
                    continue
                n_row = (row + d_row) % 4
                n_col = (col + d_col) % 4
                alive += snap[n_row * 4 + n_col]

        # --- CRITICAL CHANGE: Explicitly update 'flux1_grid' for all cases ---
        if step_due:
            if point == 1:
                flux2_grid[idx] = 1 if alive in (2, 3) else 0
            else:
                flux2_grid[idx] = 1 if alive == 3 else 0
'''
